# 第 5 章：舊帳無痛遷移與資料備份

## 1. SettleUp CSV 解析器實作

為了讓使用者從舊系統（如 SettleUp）無痛遷移，系統內建智慧解析器，可自動處理多付款人分號格式、多幣別與異常標籤過濾：

```javascript
/**
 * 解析 SettleUp 試算表的一列資料
 * @param {Array} row - 試算表單列資料
 * @param {number} rowNum - 列號（排錯使用）
 * @param {string} myName - 使用者在 SettleUp 中的顯示名稱
 */
function parseSettleUpSheetRow(row, rowNum, myName) {
  const whoPaid = String(row[0] || '').trim();
  const amountRaw = String(row[1] || '').trim();
  const forWhom = String(row[3] || '').trim();
  const splitAmounts = String(row[4] || '').trim();
  const purpose = String(row[5] || '支出').trim();
  const category = String(row[6] || '').trim() || autoDetectCategory(purpose);
  const dateTime = row[7];
  const type = String(row[10] || '').trim();

  // 1. 金額解析：處理 "178;230" 複合多付款人字串
  let amount = 0;
  let actualPayments = {};

  if (amountRaw.includes(';')) {
    const amounts = amountRaw.split(';').map(a => parseFloat(a.trim()) || 0);
    amount = amounts.reduce((sum, a) => sum + a, 0);

    if (whoPaid.includes(';')) {
      const payers = whoPaid.split(';').map(p => p.trim());
      for (let i = 0; i < payers.length && i < amounts.length; i++) {
        actualPayments[payers[i]] = amounts[i];
      }
    }
  } else {
    amount = parseFloat(amountRaw) || 0;
  }

  // 2. 債務結算 (transfer) 自動轉換為結算記錄
  if (type === 'transfer') {
    return {
      date: dateTime,
      item: '[💰結算] ' + purpose,
      amount: amount,
      originalAmount: amount,
      currency: 'TWD',
      exchangeRate: 1,
      payer: (whoPaid === myName) ? '你' : '對方',
      actualPayer: (whoPaid === myName) ? '你' : '對方',
      yourPart: 0,
      partnerPart: 0,
      yourActualPaid: 0,
      partnerActualPaid: 0,
      category: '結算',
      recordType: 'settlement',
      type: type
    };
  }

  // 3. 一般支出：解析責任分攤資訊
  const splitInfo = parseSplitInfo(forWhom, splitAmounts, amount, whoPaid, myName, actualPayments);

  return {
    date: dateTime,
    item: purpose,
    amount: amount,
    originalAmount: amount,
    currency: 'TWD',
    exchangeRate: 1,
    category: category,
    payer: (whoPaid === myName) ? '你' : '對方',
    actualPayer: (whoPaid === myName) ? '你' : '對方',
    yourPart: splitInfo.yourPart,
    partnerPart: splitInfo.partnerPart,
    yourActualPaid: splitInfo.yourActualPaid || 0,
    partnerActualPaid: splitInfo.partnerActualPaid || 0,
    paymentAccount: '',
    project: '',
    recordType: 'expense',
    type: type
  };
}
```

## 2. AndroMoney 舊帳解析與分類映射矩陣

系統內建 `mapAndroMoneyCategory` 字典，將 AndroMoney 雙層分類對齊至系統 7 大標準主分類（`飲食`、`居住`、`交通`、`娛樂`、`寵物`、`服飾`、`其他`）：

```javascript
function mapAndroMoneyCategory(category, subCategory) {
  const target = subCategory || category;
  const categoryMap = {
    // 飲食
    '早餐': '飲食', '午餐': '飲食', '晚餐': '飲食', '飲料': '飲食', '點心零嘴': '飲食', '食材': '飲食',
    // 交通
    '交通': '交通', '停車費': '交通', '計程車': '交通', '公車': '交通', '油錢': '交通', '捷運': '交通', '高鐵': '交通',
    // 居住
    '房租': '居住', '水費': '居住', '電費': '居住', '瓦斯': '居住', '網路': '居住', '日用品': '居住',
    // 娛樂
    '休閒娛樂': '娛樂', '電影': '娛樂', '旅遊': '娛樂', '運動': '娛樂', '遊戲': '娛樂',
    // 收入
    '薪資': '收入', '獎金': '收入', '兼差': '收入', '打工': '收入', '利息': '收入'
  };

  return categoryMap[target] || category || '其他';
}
```

## 3. 試算表定期備份與操作日誌工作表

為防止誤刪或協作者操作失誤，系統內建操作日誌工作表 (`logAction`)，並透過 Google 雲端硬碟 API 實施唯讀保護：

```javascript
/**
 * 記錄操作日誌（敏感資訊脫敏與防篡改保護）
 */
function logAction(action, details) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let logSheet = ss.getSheetByName('操作日誌');

  if (!logSheet) {
    logSheet = ss.insertSheet('操作日誌');
    logSheet.getRange(1, 1, 1, 5).setValues([['時間', '使用者', '動作', '詳細資訊', '裝置識別']]);
    logSheet.getRange(1, 1, 1, 5)
      .setBackground(CONFIG.COLORS.HEADER)
      .setFontColor('#ffffff')
      .setFontWeight('bold');

    // 鎖定工作表：僅試算表擁有者具備編輯權限
    const protection = logSheet.protect().setDescription('操作日誌防篡改保護');
    protection.removeEditors(protection.getEditors());
    protection.addEditor(ss.getOwner().getEmail());
  }

  logSheet.appendRow([
    Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd HH:mm:ss'),
    Session.getActiveUser().getEmail(),
    action,
    escapeHtml(String(details)),
    Session.getTemporaryActiveUserKey()
  ]);
}
```
